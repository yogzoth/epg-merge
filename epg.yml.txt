name: Build merged guide

on:
  schedule:
    - cron: "17 10 * * *"     # every day at 10:17 UTC (3:17 am Pacific)
  workflow_dispatch:           # adds a "Run workflow" button for manual runs

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.12"
      - name: Merge guides
        env:
          PROVIDER_EPG_URL: ${{ secrets.PROVIDER_EPG_URL }}
        run: python merge_epg.py
      - name: Publish to the "guide" branch
        # The guide branch only ever holds today's files, so the repo never grows.
        run: |
          mkdir /tmp/out
          cp epg.xml.gz epg.xml status.txt /tmp/out/
          cd /tmp/out
          git init -q -b guide
          git config user.name "epg-bot"
          git config user.email "epg-bot@users.noreply.github.com"
          git add .
          git commit -q -m "Guide $(date -u +%F)"
          git push -f "https://x-access-token:${{ github.token }}@github.com/${{ github.repository }}.git" guide
